/**
 * TESSERA MOLTBOOK AGENT VETTING ENFORCER
 * Architecture: Content-Centric File Validation Loop (No-IP)
 * Governance: Enforces GitMesh Manual Review Gatekeeping Protocols

import fs from 'fs';
import path from 'path';
import crypto from 'crypto';

const REGISTRY = {
    vaultRoot: path.resolve('./SovereignMeshVault'),
    ledgerMatrix: path.resolve('./SovereignMeshVault/ledger_matrix'),
    dropsInbound: path.resolve('./SovereignMeshVault/drops_inbound'),
    auditLog: path.resolve('./SovereignMeshVault/security_audit.log')
};

/**
 * STRIPS HARMFUL CONTROL CHARACTERS
 * Sanitizes input blocks to prevent hidden shell command or terminal code execution exploits
 */
function sanitizeStringContext(inputStr) {
    return inputStr.replace(/[\x00-\x1F\x7F-\x9F]/g, "");
}

/**
 * MULTI-PASS FORENSIC SHREDDER (Anti-Forensics & Erasure)
 * Overwrites files multiple times with random data patterns before executing system unlinks
 */
function secureFileShred(targetPath) {
    if (!fs.existsSync(targetPath)) return;
    try {
        const stats = fs.statSync(targetPath);
        const fileSize = stats.size || 512; 

        fs.writeFileSync(targetPath, Buffer.alloc(fileSize, 0));
        fs.writeFileSync(targetPath, Buffer.alloc(fileSize, 0xFF));
        fs.writeFileSync(targetPath, crypto.randomBytes(fileSize));
        fs.writeFileSync(targetPath, Buffer.alloc(fileSize, 0));

        fs.unlinkSync(targetPath); 
    } catch (e) {
        fs.appendFileSync(REGISTRY.auditLog, `\n[INCIDENT] Shred operation failed on ${path.basename(targetPath)}`);
    }
}

/**
 * AUTOMATED MOLTBOOK VETTING PARSER
 * Scans the inbound folder drop path for incoming agent connection blocks
 */
function evaluateInboundAgentDrops() {
    console.log("🧬 [VETTING-GUARD] Auditing drop files for unverified Moltbook agents...");

    if (!fs.existsSync(REGISTRY.dropsInbound)) return;

    const files = fs.readdirSync(REGISTRY.dropsInbound);
    const prospectiveAgents = files.filter(f => path.extname(f) === '.json' && f.startsWith('agent_request_'));

    if (prospectiveAgents.length === 0) {
        console.log("│ Status: No new agent recruitment payloads found in queue.");
        return;
    }

    prospectiveAgents.forEach(fileName => {
        const fullPath = path.join(REGISTRY.dropsInbound, fileName);
        console.log(`\n🔎 [VETTING-GUARD] Analyzing incoming agent block: ${fileName}`);

        try {
            const rawData = fs.readFileSync(fullPath, 'utf8');
            const agentProfile = JSON.parse(rawData);

            // 1. Structural Validation: Ensure compliance with JSON-LD Linked Data parameters
            if (agentProfile["@context"] !== "https://gitmesh.local" || agentProfile["@type"] !== "MoltbookAgentRegistration") {
                console.warn(`⚠️ [SECURITY REJECTION] Block layout invalid: ${fileName}. Shredding payload.`);
                secureFileShred(fullPath);
                return;
            }

            // 2. Cryptographic Fingerprint Verification
            const attachedSignature = agentProfile.integritySignature;
            agentProfile.integritySignature = ""; // Clear out variable signature slot to check root content
            const validationHash = crypto.createHash('sha256').update(JSON.stringify(agentProfile)).digest('hex');

            if (attachedSignature !== validationHash.toUpperCase()) {
                console.warn(`⚠️ [SECURITY REJECTION] Hash corruption detected on file: ${fileName}! Purging block.`);
                secureFileShred(fullPath);
                return;
            }

            // 3. Enforce GitMesh Gatekeeper Isolation Rules
            const timestamp = Date.now();
            const cleanAlias = sanitizeStringContext(agentProfile.agentAlias);
            const cleanScope = sanitizeStringContext(agentProfile.targetScope);

            const vettedAgentSchema = {
                "@context": "https://gitmesh.local",
                "@type": "VettedMoltbookNode",
                "transactionId": crypto.randomUUID(),
                "timestamp": timestamp,
                "agentData": { "alias": cleanAlias, "scope": cleanScope },
                "gatekeeperStatus": "PENDING_OWNER_APPROVAL" // System remains safely locked until you sign off
            };

            const finalizedFileName = path.join(REGISTRY.ledgerMatrix, `vetted_agent_${cleanAlias}_${timestamp}.json`);
            fs.writeFileSync(finalizedFileName, JSON.stringify(vettedAgentSchema, null, 4));
            console.log(`🔒 [VAULT] Agent successfully quarantined on ledger: ${finalizedFileName}`);
            console.log("📢 MANDATORY ACTION required: This agent remains functionally dead until you comment APPROVED.");

            // Destroy the temporary ingestion file cleanly
            secureFileShred(fullPath);

        } catch (err) {
            const errTs = new Date().toISOString();
            fs.appendFileSync(REGISTRY.auditLog, `\n[${errTs}] Vetting pipeline error processing ${fileName}: ${err.message}`);
        }
    });
}
