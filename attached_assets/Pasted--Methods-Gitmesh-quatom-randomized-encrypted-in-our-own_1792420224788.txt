import fs from 'fs';
import { execSync } from 'child_process';

// 1. Fetch live data while Starlink is active

// 2. Format it cleanly as text for your offline agent to read

fs.writeFileSync(filePath, content);

// 4. Track changes via Git without opening a browser
execSync('git add ../../mesh/drops/live_feed.md');
execSync(`git commit -m "Auto-drop: Web update captured at ${timestamp}"`);
console.log("Web data successfully converted into an offline Git file.");
